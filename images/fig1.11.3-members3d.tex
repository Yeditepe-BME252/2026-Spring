\documentclass[tikz, border=14pt]{standalone}
\usepackage{tikz}
\usepackage{tikz-3dplot}
\usepackage{amsmath}
\usetikzlibrary{arrows.meta, calc}

%% ── Colours ──────────────────────────────────────────────────
\definecolor{mfill} {RGB}{200,218,240}
\definecolor{mfillD}{RGB}{155,182,215}
\definecolor{mfillT}{RGB}{228,238,250}
\definecolor{mline} {RGB}{ 40, 70,130}
\definecolor{cfill} {RGB}{255,238,185}
\definecolor{cline} {RGB}{148, 98,  0}
\definecolor{dcol}  {RGB}{148,  0,  0}
\definecolor{acol}  {RGB}{ 10,110, 10}
\definecolor{hfill} {RGB}{240,242,245}

\tikzset{
  ME/.style={draw=mline, line width=0.85pt},
  CE/.style={draw=cline, line width=1.10pt},
  DA/.style={dcol,
    {Latex[length=3pt,width=2pt]}-{Latex[length=3pt,width=2pt]},
    line width=0.60pt},
  DE/.style={dcol, line width=0.45pt},
  AX/.style={acol, line width=0.65pt,
    dash pattern=on 5.5pt off 1.8pt on 1.2pt off 1.8pt},
}

%% ── View: elevation 72 deg, azimuth 115 deg ─────────────────
\tdplotsetmaincoords{70}{110}

\begin{document}
\begin{tikzpicture}[tdplot_main_coords]

%%=============================================================
%% GLOBAL PARAMETERS
%%=============================================================
\pgfmathsetmacro{\LL}{7.5}    % member length (x-axis)
\pgfmathsetmacro{\MX}{\LL/2}  % mid-span x (cut plane)
\pgfmathsetmacro{\SEP}{3.8}   % lateral spacing (y-axis)

%%=============================================================
%% MEMBER 1 – RECTANGULAR  (half-width W in z, half-thick T in y)
%% Longitudinal axis along x, centred at y=0, z=0
%%=============================================================
\pgfmathsetmacro{\W}{0.65}   % half-width  (z)
\pgfmathsetmacro{\T}{0.45}   % half-thickness (y)
\pgfmathsetmacro{\YA}{0}     % y-centre member 1

% Back face (x=0)
\fill[mfill!90, ME]
  (0,-\T,-\W)--(0,\T,-\W)--(0,\T,\W)--(0,-\T,\W)--cycle;

% Bottom face (z=-W)
\fill[mfillD!75, ME]
  (0,-\T,-\W)--(\LL,-\T,-\W)--(\LL,\T,-\W)--(0,\T,-\W)--cycle;

% Right side face (y=+T) – visible
\fill[mfill, ME, opacity=0.52]
  (0,\T,-\W)--(\LL,\T,-\W)--(\LL,\T,\W)--(0,\T,\W)--cycle;

% Cut plane at x=MX
\fill[cfill, CE, opacity=0.92]
  (\MX,-\T,-\W)--(\MX,\T,-\W)--(\MX,\T,\W)--(\MX,-\T,\W)--cycle;

% Top face (z=+W)
\fill[mfillT, ME, opacity=0.52]
  (0,-\T,\W)--(\LL,-\T,\W)--(\LL,\T,\W)--(0,\T,\W)--cycle;

% Front face (x=LL)
\fill[mfillD, ME, opacity=0.7]
  (\LL,-\T,-\W)--(\LL, \T,-\W)--(\LL, \T,\W)--(\LL,-\T,\W)--cycle;

% Axis
\draw[AX] (-0.55,\YA,0)--(\LL+0.65,\YA,0);

% Dim w (full width = 2W, in z-dir on cut face)
\pgfmathsetmacro{\Toff}{\T+0.22}
\draw[DA] (\MX,\Toff,-\W)--(\MX,\Toff,\W);
\node[dcol, font=\small] at (\MX,\Toff+0.05,0) [below right] {$w$};

% Dim L
\pgfmathsetmacro{\Lbot}{\W+0.25}
\draw[DE] (0,-\T,\Lbot)--(0,-\T,\Lbot+0.22);
\draw[DE] (\LL,-\T,\Lbot)--(\LL,-\T,\Lbot+0.22);
\draw[DA] (0,-\T,\Lbot+0.14)--(\LL,-\T,\Lbot+0.14);
\node[dcol, font=\small] at (\MX,-\T,\Lbot+0.14) [above] {$L$};

% Dim t (full thickness = 2T, in y-dir on bottom)
\pgfmathsetmacro{\Woff}{\W+0.22}
\draw[DA] (\MX,-\T,\Woff+.08)--(\MX,\T,\Woff+.08);
\node[dcol, font=\small] at (\MX,0,\Woff+.08) [above right] {$t$};

\node[mline, font=\footnotesize\bfseries]
  at (\MX,\YA,-\W-1.42) {Rectangular};

%%=============================================================
%% MEMBER 2 – SOLID CIRCULAR  (radius Ro)
%% Centre at y=SEP
%%=============================================================
\pgfmathsetmacro{\Ro}{0.75}
\pgfmathsetmacro{\TH}{136}
\pgfmathsetmacro{\YB}{\SEP}

% Back ellipse (x=0)
\fill[mfill]
  plot[domain=0:360, samples=72, smooth]
  ({0}, {\YB + \Ro*cos(\x)}, {\Ro*sin(\x)}) -- cycle;
\draw[mline, line width=0.85pt]
  plot[domain=0:360, samples=72, smooth]
  ({0}, {\YB + \Ro*cos(\x)}, {\Ro*sin(\x)});

% Cut ellipse at x=MX
\fill[cfill, opacity=0.92]
  plot[domain=0:360, samples=72, smooth]
  ({\MX}, {\YB + \Ro*cos(\x)}, {\Ro*sin(\x)}) -- cycle;
\draw[cline, line width=1.1pt]
  plot[domain=0:360, samples=72, smooth]
  ({\MX}, {\YB + \Ro*cos(\x)}, {\Ro*sin(\x)});

% Body fill (silhouette rectangle between top and bottom tangent lines)
\fill[mfill!98, opacity=0.5]
  (0, {\YB + \Ro*cos(\TH)}, {\Ro*sin(\TH)}) -- (\LL, {\YB + \Ro*cos(\TH)}, {\Ro*sin(\TH)}) -- (\LL, {\YB + \Ro*cos(\TH+180)}, {\Ro*sin(\TH+180)}) -- (0, {\YB + \Ro*cos(\TH+180)}, {\Ro*sin(\TH+180)}) -- cycle;
\draw[mline, line width=0.85pt] (0, {\YB + \Ro*cos(\TH)}, {\Ro*sin(\TH)}) -- (\LL, {\YB + \Ro*cos(\TH)}, {\Ro*sin(\TH)});
\draw[mline, line width=0.85pt] (\LL, {\YB + \Ro*cos(\TH+180)}, {\Ro*sin(\TH+180)}) -- (0, {\YB + \Ro*cos(\TH+180)}, {\Ro*sin(\TH+180)});

  % Front ellipse (x=LL) — drawn first
\fill[mfillD, opacity=0.7]
  plot[domain=0:360, samples=72, smooth]
  ({\LL}, {\YB + \Ro*cos(\x)}, {\Ro*sin(\x)}) -- cycle;
\draw[mline, line width=0.85pt]
  plot[domain=0:360, samples=72, smooth]
  ({\LL}, {\YB + \Ro*cos(\x)}, {\Ro*sin(\x)});


% Axis
\draw[AX] (-0.55,\YB,0)--(\LL+0.65,\YB,0);

% Dim r_o  (upper-left direction in yz-plane)
\pgfmathsetmacro{\rox}{\Ro*0.50}
\pgfmathsetmacro{\roz}{\Ro*0.87}
\draw[DA] (\MX,\YB,0)--(\MX,{\YB+\rox},{\roz});
\node[dcol, font=\small] at (\MX,{\YB+\rox*0.5},{\roz}) [above right] {$r_o$};

% Dim L
\pgfmathsetmacro{\Lbot}{\Ro+0.35}
\draw[DE] (0,\YB,\Lbot-0.2)--(0,\YB,\Lbot+0.2);
\draw[DE] (\LL,\YB,\Lbot-0.2)--(\LL,\YB,\Lbot+0.2);
\draw[DA] (0,\YB,\Lbot+0.1)--(\LL,\YB,\Lbot+0.1);
\node[dcol, font=\small] at (\MX,\YB,\Lbot+0.14) [above] {$L$};

\node[mline, font=\footnotesize\bfseries]
  at (\MX,\YB,-\Ro-1.42) {Solid Circular};

%%=============================================================
%% MEMBER 3 – HOLLOW CIRCULAR  (Ri inner, Ro outer)
%% Centre at y=2*SEP
%%=============================================================
\pgfmathsetmacro{\Roo}{0.75}
\pgfmathsetmacro{\Ri}{0.42}
\pgfmathsetmacro{\YC}{2*\SEP}

% Back outer ellipse
\fill[mfill]
  plot[domain=0:360, samples=72, smooth]
  ({0}, {\YC + \Roo*cos(\x)}, {\Roo*sin(\x)}) -- cycle;
\draw[mline, line width=0.85pt]
  plot[domain=0:360, samples=72, smooth]
  ({0}, {\YC + \Roo*cos(\x)}, {\Roo*sin(\x)});
% Back inner hole
%\fill[hfill]
%  plot[domain=0:360, samples=72, smooth]
%  ({0}, {\YC + \Ri*cos(\x)}, {\Ri*sin(\x)}) -- cycle;
%\draw[mline, dashed, line width=0.85pt]
%  plot[domain=0:360, samples=72, smooth]
%  ({0}, {\YC + \Ri*cos(\x)}, {\Ri*sin(\x)});

% moved back inner hole above the body silhouette
\draw[mline, dashed, line width=0.85pt]
  plot[domain=0:360, samples=72, smooth]
  ({0}, {\YC + \Ri*cos(\x)}, {\Ri*sin(\x)});

  % Cut annular face
\fill[cfill, opacity=0.92]
  plot[domain=0:360, samples=72, smooth]
  ({\MX}, {\YC + \Roo*cos(\x)}, {\Roo*sin(\x)}) -- cycle;
%\fill[hfill]
%  plot[domain=0:360, samples=72, smooth]
%  ({\MX}, {\YC + \Ri*cos(\x)}, {\Ri*sin(\x)}) -- cycle;
\draw[cline, line width=1.1pt]
  plot[domain=0:360, samples=72, smooth]
  ({\MX}, {\YC + \Roo*cos(\x)}, {\Roo*sin(\x)});
\draw[cline, line width=1.1pt]
  plot[domain=0:360, samples=72, smooth]
  ({\MX}, {\YC + \Ri*cos(\x)}, {\Ri*sin(\x)});

% Body silhouette
\fill[mfill!80, opacity=.5]
  (0, {\YC + \Ro*cos(\TH)}, {\Ro*sin(\TH)}) -- (\LL, {\YC + \Ro*cos(\TH)}, {\Ro*sin(\TH)}) -- (\LL, {\YC + \Ro*cos(\TH+180)}, {\Ro*sin(\TH+180)}) -- (0, {\YC + \Ro*cos(\TH+180)}, {\Ro*sin(\TH+180)}) -- cycle;
\draw[mline, line width=0.85pt] (0, {\YC + \Ro*cos(\TH)}, {\Ro*sin(\TH)}) -- (\LL, {\YC + \Ro*cos(\TH)}, {\Ro*sin(\TH)});
\draw[mline, line width=0.85pt] (\LL, {\YC + \Ro*cos(\TH+180)}, {\Ro*sin(\TH+180)}) -- (0, {\YC + \Ro*cos(\TH+180)}, {\Ro*sin(\TH+180)});

% Front outer ellipse
\fill[mfillD, opacity=0.7]
  plot[domain=0:360, samples=72, smooth]
  ({\LL}, {\YC + \Roo*cos(\x)}, {\Roo*sin(\x)}) -- cycle;
\draw[mline, line width=0.85pt]
  plot[domain=0:360, samples=72, smooth]
  ({\LL}, {\YC + \Roo*cos(\x)}, {\Roo*sin(\x)});
% Front inner hole
\fill[mfillD!140]
  plot[domain=0:360, samples=72, smooth]
  ({\LL}, {\YC + \Ri*cos(\x)}, {\Ri*sin(\x)}) -- cycle;
\draw[mline, line width=0.85pt]
  plot[domain=0:360, samples=72, smooth]
  ({\LL}, {\YC + \Ri*cos(\x)}, {\Ri*sin(\x)});

% Axis
\draw[AX] (-0.55,\YC,0)--(\LL+0.65,\YC,0);

% Dim r_o  (upper-left)
\pgfmathsetmacro{\rox}{\Roo*0.50}
\pgfmathsetmacro{\roz}{\Roo*0.87}
\draw[DA] (\MX,\YC,0)--(\MX,{\YC+\rox},{\roz});
\node[dcol, font=\small] at (\MX,{\YC+\rox*0.5},{\roz}) [above right] {$r_o$};

% Dim r_i  (lower-left)
\pgfmathsetmacro{\rix}{\Ri*0.50}
\pgfmathsetmacro{\riz}{\Ri*0.87}
\draw[DA] (\MX,\YC,0)--(\MX,{\YC+\rix},{-\riz});
\node[dcol, font=\small] at (\MX,{\YC+\rix*0.5},{-\riz}) [below right] {$r_i$};

% Dim L
\pgfmathsetmacro{\Lbot}{\Ro+0.35}
\draw[DE] (0,\YC,\Lbot-0.2)--(0,\YC,\Lbot+0.2);
\draw[DE] (\LL,\YC,\Lbot-0.2)--(\LL,\YC,\Lbot+0.2);
\draw[DA] (0,\YC,\Lbot+0.1)--(\LL,\YC,\Lbot+0.1);
\node[dcol, font=\small] at (\MX,\YC,\Lbot+0.14) [above] {$L$};

\node[mline, font=\footnotesize\bfseries]
  at (\MX,\YC,-\Roo-1.42) {Hollow Circular};

\end{tikzpicture}
